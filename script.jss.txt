function getCart() {
  try { return JSON.parse(localStorage.getItem("cart")) || []; }
  catch (e) { return []; }
}

function saveCart(cart) {
  localStorage.setItem("cart", JSON.stringify(cart));
  updateCartCount();
}

function updateCartCount() {
  var el = document.getElementById("cart-count");
  if (!el) return;
  var count = 0;
  getCart().forEach(function (i) { count += i.qty; });
  el.textContent = count;
}

function addToCart(btn) {
  var p = btn.closest(".product");
  var priceText = p.querySelector(".price").innerText.trim();
  var img = p.querySelector("img");
  var item = {
    name: p.querySelector("h3").innerText.trim(),
    price: parseFloat(priceText.replace(/[^0-9.]/g, "")) || 0,
    symbol: priceText.indexOf("$") !== -1 ? "$" : "₹",
    img: img ? img.getAttribute("src") : "",
    qty: 1
  };
  var cart = getCart();
  var found = cart.find(function (c) { return c.name === item.name; });
  if (found) { found.qty++; } else { cart.push(item); }
  saveCart(cart);
  btn.textContent = "Added ✓";
  setTimeout(function () { btn.textContent = "Add to Cart"; }, 1000);
}

function changeQty(index, change) {
  var cart = getCart();
  cart[index].qty += change;
  if (cart[index].qty <= 0) cart.splice(index, 1);
  saveCart(cart);
  renderCart();
  fillOrderItems();
}

function removeItem(index) {
  var cart = getCart();
  cart.splice(index, 1);
  saveCart(cart);
  renderCart();
  fillOrderItems();
}

function clearCart() {
  saveCart([]);
  renderCart();
  fillOrderItems();
}

function renderCart() {
  var box = document.getElementById("cart-items");
  var totalBox = document.getElementById("cart-total");
  if (!box) return;

  var cart = getCart();
  if (cart.length === 0) {
    box.innerHTML = "<p>Your cart is empty.</p>";
    totalBox.innerHTML = "";
    return;
  }

  var html = "";
  var totals = {};
  cart.forEach(function (item, i) {
    totals[item.symbol] = (totals[item.symbol] || 0) + item.price * item.qty;
    html +=
      '<div class="cart-row">' +
        '<img src="' + item.img + '" alt="" onerror="this.style.display=\'none\'">' +
        '<div class="cart-info"><h3>' + item.name + '</h3>' +
        '<p>' + item.symbol + item.price.toLocaleString("en-IN") + '</p></div>' +
        '<div class="cart-qty">' +
          '<button onclick="changeQty(' + i + ', -1)">-</button>' +
          '<span>' + item.qty + '</span>' +
          '<button onclick="changeQty(' + i + ', 1)">+</button>' +
        '</div>' +
        '<button class="remove-btn" onclick="removeItem(' + i + ')">Remove</button>' +
      '</div>';
  });
  box.innerHTML = html;

  var totalHtml = "<h3>Total</h3>";
  for (var sym in totals) {
    totalHtml += "<p>" + sym + totals[sym].toLocaleString("en-IN") + "</p>";
  }
  totalBox.innerHTML = totalHtml;
}

function fillOrderItems() {
  var form = document.getElementById("orderForm");
  if (!form) return;
  var box = form.querySelector("#item");
  box.value = getCart().map(function (c) {
    return c.name + " x" + c.qty;
  }).join(", ");
}

function searchItems() {
  var text = document.getElementById("search").value.toLowerCase();
  var items = document.getElementsByClassName("product");
  for (var i = 0; i < items.length; i++) {
    var h3 = items[i].querySelector("h3");
    var name = h3 ? h3.innerText.toLowerCase() : "";
    items[i].style.display = name.includes(text) ? "block" : "none";
  }
}

function showResult(form, text, ok) {
  var box = form.querySelector(".result");
  if (!box) { alert(text); return; }
  box.textContent = text;
  box.style.color = ok ? "green" : "red";
}
